from flask import Flask, render_template, request, redirect
import sqlite3

app = Flask(__name__)

def init_db():
    conn = sqlite3.connect("database.db")
    c = conn.cursor()
    c.execute("""
        CREATE TABLE IF NOT EXISTS movimentacoes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            tipo TEXT,
            descricao TEXT,
            valor REAL
        )
    """)
    conn.commit()
    conn.close()

@app.route("/")
def index():
    conn = sqlite3.connect("database.db")
    c = conn.cursor()
    c.execute("SELECT * FROM movimentacoes")
    dados = c.fetchall()
    conn.close()
    saldo = sum([d[3] if d[1] != "saida" else -d[3] for d in dados])
    return render_template("index.html", dados=dados, saldo=saldo)

@app.route("/add", methods=["POST"])
def add():
    tipo = request.form["tipo"]
    descricao = request.form["descricao"]
    valor = float(request.form["valor"])
    conn = sqlite3.connect("database.db")
    c = conn.cursor()
    c.execute("INSERT INTO movimentacoes (tipo, descricao, valor) VALUES (?, ?, ?)", (tipo, descricao, valor))
    conn.commit()
    conn.close()
    return redirect("/")

if __name__ == "__main__":
    init_db()
    app.run(host="0.0.0.0", port=5000)
